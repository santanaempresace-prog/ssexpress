rules_version = '2';

service cloud.firestore {
  match /databases/{database}/documents {

    // FINANCEIRO
    // Pode VER: Caio e Edilson
    function leitor() {
      return request.auth != null
        && request.auth.token.email in [
          'santana.empresace@gmail.com',
          'esposo.santanapreto@gmail.com'
        ];
    }
    // Pode LANÇAR e ALTERAR: só o Caio
    function editor() {
      return request.auth != null
        && request.auth.token.email == 'santana.empresace@gmail.com';
    }
    match /fin_lanc/{id} { allow read: if leitor(); allow write: if editor(); }
    match /fin_cfg/{id}  { allow read: if leitor(); allow write: if editor(); }

    // RASTREIO: continua aberto, igual a hoje
    match /cargotrack/{doc=**}            { allow read, write: if true; }
    match /cargotrack_ctes/{doc=**}       { allow read, write: if true; }
    match /cargotrack_eventos/{doc=**}    { allow read, write: if true; }
    match /cargotrack_nfechaves/{doc=**}  { allow read, write: if true; }
    match /cargotrack_obs_photos/{doc=**} { allow read, write: if true; }
    match /cargotrack_photos/{doc=**}     { allow read, write: if true; }
    match /cargotrack_xmls/{doc=**}       { allow read, write: if true; }
  }
}
